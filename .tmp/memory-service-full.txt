 201|function score(query: string, item: Info) {
 202|  const terms = query
 203|    .toLowerCase()
 204|    .split(/\s+/)
 205|    .map((term) => term.trim())
 206|    .filter(Boolean)
 207|  if (!terms.length) return item.importance
 208|  const haystack = [item.content, item.summary ?? "", ...item.tags].join(" ").toLowerCase()
 209|  const hits = terms.reduce((count, term) => count + (haystack.includes(term) ? 1 : 0), 0)
 210|  return hits * 10 + item.importance

 213|function explicitMemory(text: string) {
 214|  const patterns = [
 215|    /(?:please\s+)?remember(?:\s+that)?\s+(.+)/i,
 216|    /(?:请记住|记住)[:：]?\s*(.+)/,
 217|    /帮我记住[:：]?\s*(.+)/,
 218|    /以后都(?:要|用|按|记住)\s*(.+)/,
 219|    /我(?:的)?偏好(?:是|：|:)\s*(.+)/,
 220|    /从现在起\s*(.+)/,
 221|    /务必记住\s*(.+)/,
 222|  ]
 223|  for (const pattern of patterns) {
 224|    const match = text.match(pattern)
 225|    const content = match?.[1]?.trim()
 226|    if (content && content.length >= 4) return content
 227|  }

 316|  const root = path.join(location.worktree === "/" ? location.directory : location.worktree, ".novaway", "memory")

 582|        ? yield* listReviewCandidates({
